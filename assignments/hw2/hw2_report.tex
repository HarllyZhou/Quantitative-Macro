\section*{Question 1}
\begin{enumerate}[label=\alph*.]
   \item Taking logarithm, we have
   \begin{align*}
      \log A_t = z_t + \alpha \log \Gamma_t.
   \end{align*}
   Taking difference, we get
   \begin{align*}
      \Delta a_t = \log A_t - \log A_{t-1} =  (\rho_z - 1) z_{t-1} + \alpha g_t + e_t^z.
   \end{align*}
   Since $\{z_t\}$ and $\{g_t\}$ are stationary and $e_t^z \overset{i.i.d.}{\sim} \mathcal{N}(0, \sigma_z^2)$, we have $\Delta a_t$ being stationary. So $d=1$.

   Note that
   \begin{align*}
      (1-\rho_z L) \Delta z_t = (1-L) e_t^z, \text{ and }\, (1-\rho_g L) g_t = (1-\rho_g) \mu_g + e_t^g.
   \end{align*}
   Multiplying $\Delta a_t = \Delta z_t + \alpha \Delta g_t$ by $(1-\rho_z L)(1-\rho_g L)$, we get
   \begin{align}\label{eq:arima(2,1,2)}
      (1-\rho_z L)(1-\rho_g L) \Delta a_t = (1-\rho_g L) (1-L) e_t^z + \alpha (1-\rho_z L)[(1-\rho_g)\mu_g + e_t^g].
   \end{align}
   The LHS characterizes an $AR(2)$ process and the RHS characterizes a $MA(2)$ process. So $p=2, q=2$.

   % \item By stability and invertibility conditions, we can expand the $ARIMA(2,1,2)$ process as
   % \begin{align*}

   % \end{align*}
   \newpage

   \item Consider the stationary $z$ part first. We have
   \begin{align*}
      z_{t+h} = \sum_{j=0}^{+\infty} \rho_z^j e_{t+h-j}^z.
   \end{align*}
   Given $\{z_\tau\}_{\tau=-\infty}^{t-1}$, the forecase error variance is
   \begin{align*}
      FEV_h^z = \operatorname{Var}\left[\sum_{j=0}^h \rho_z^j e_{t+h-j}^z\right] = \sigma_z^2 \frac{1-\rho_z^{2(h+1)}}{1-\rho_z^2}.
   \end{align*}

   Now consider the trend component $\log \Gamma_t$. Note that
   \begin{align*}
      \log \Gamma_{t+h} = \log \Gamma_{t-1} + \sum_{j=0}^h g_{t+j}.
   \end{align*}
   Ignoring the constant term, the $g_t$ process can be written as
   \begin{align*}
      g_{t+h} = \sum_{j=0}^{+\infty} \rho_g^j e_{t+h-j}^g.
   \end{align*}
   Given $\{z_\tau\}_{\tau=-\infty}^{t-1}$, the forecase error variance is
   \begin{align*}
      FEV_h^\Gamma = \operatorname{Var}\left[\sum_{j=0}^h \sum_{i=0}^j \rho_g^i e_{t+j-i}^g \right].
   \end{align*}
   We can compute the values by plugging in the parameter values. Figure \ref{fig:q1b} shows that plot. 
   \begin{figure}[htbp]
      \centering
      \includegraphics[width=0.6\textwidth]{hw2_q1b.png}
      \caption{Forecast Error Variance}
      \label{fig:q1b}
   \end{figure}
    
   \newpage
   \item The decomposition is shown in Figure \ref{fig:q1c}.
   \begin{figure}[htbp]
      \centering
      \includegraphics[width=0.6\textwidth]{hw2_q1c.png}
      \caption{Decomposition of Forecast Error Variance}
      \label{fig:q1c}
   \end{figure}

   The effect of transitory components decays with time. The permanent components have a more persistent effect. From the expressions of $FEV^z$ and $FEV^{\Gamma}$, mathematically, we can observe that each term of $e^z$ is summed once, while each term of $e^g$ is summed multiple times. This makes the effect from permanent components accumulate.
\end{enumerate}

\begin{enumerate}[label=\alph*.]
   \item The expectation is
   \begin{align*}
      \mathbb{E}[x_t] &= \mathbb{E}[\alpha] \cos t + \mathbb{E}[\beta] \sin t\\
      & = 0.
   \end{align*}
   The variance is
   \begin{align*}
      \operatorname{Var}[x_t] &= \operatorname{Var}[\alpha] \cos^2 t + \operatorname{Var}[\beta] \sin^2 t + 2 \operatorname{Cov}(\alpha, \beta) \cos t \sin t\\
      & = \cos^2 t + \sin^2 t\\
      & = 1.
   \end{align*}
   The second line is by the independence of $\alpha$ and $\beta$.

   \item No. Note that
   \begin{align*}
      \mathbb{E}[\alpha^2] &= \operatorname{Var}[\alpha] + \left(\mathbb{E}[\alpha]\right)^2\\
      &= 1\\
      \mathbb{E}[\beta^2] &= \operatorname{Var}[\beta] + \left(\mathbb{E}[\beta]\right)^2\\
      &= 1\\
      \mathbb{E}[\alpha \beta] &= \operatorname{Cov}(\alpha, \beta) + \mathbb{E}[\alpha] \mathbb{E}[\beta]\\
      &= 0.
   \end{align*}
   As a result, we have
   \begin{align*}
      \mathbb{E}[x_t x_{t-k}] &= \mathbb{E}[\alpha^2] \cos t \cos (t-k) + \mathbb{E}[\beta^2] \sin t \sin (t-k)\\
      &\qquad \quad + \mathbb{E}[\alpha \beta] [\cos t \sin (t-k) + \sin t \cos (t-k)]\\
      &= \cos t \cos (t-k) + \sin t \sin (t-k)\\
      &= \cos [t-(t-k)]\\
      &= \cos k.
   \end{align*}
\end{enumerate}
