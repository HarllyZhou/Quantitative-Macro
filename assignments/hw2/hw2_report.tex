\section*{Question 2}
\begin{enumerate}[label=\alph*.]
    \item Rewriting the $ARMA(1,1)$ process, we get
    \begin{align*}
      (1-0.8L)y_t = (1+0.4L)e_t.
    \end{align*} 
    Suppose that $y_t = \psi(L)e_t$. Then we have
    \begin{align*}
      (1-0.8L)(\psi_0 + \psi_1 L + \cdots) = 1 + 0.4 L.
    \end{align*}
    Equating coefficients, we get the impulse responses:
    \[
      \psi_j = \left\{
         \begin{array}{ll}
            1, & j=0\\
            1.2 \times 0.8^{j-1}, & j \geq 1\\
         \end{array}
      \right.
    \]
    

    Plotting this function, we get Figure \ref{fig:2a}. Codes in ``hw2\_q2.R''.
    \begin{figure}[htbp]
      \centering
      \includegraphics[width=0.6\textwidth]{hw2_q2a_irf.png}
      \caption{IRF to unit shock in $e_0$}
      \label{fig:2a}
    \end{figure}


    \item We can write $y_t$ as a $MA(\infty)$ process:
    \begin{align*}
      y_t = \sum_{j=0}^{\infty} \psi_j e_{t-j}.
    \end{align*}
    
    It immediately follows that $\mathbb{E}[y_t] = 0$.

    The variance is
    \begin{align*}
      \operatorname{Var}[y_t] = \operatorname{Var}[e_t](\sum_{j=0}^{\infty} \psi_j^2) = 3\left(1+\frac{1.2^2}{1-0.64}\right) = 15.
    \end{align*}

    The autocovariances are
    \begin{align*}
      \gamma(1) &= \operatorname{Var}[e_t](\sum_{j=0}^{\infty} \psi_j \psi_{j+1}) = 3\left(1.2 + \frac{1.2^2\times0.8}{1-0.64}\right) = 13.2.\\
      \gamma(2) &= \operatorname{Var}[e_t](\sum_{j=0}^{\infty} \psi_j \psi_{j+2}) = 3\left(0.96 + \frac{1.2^2\times0.8^2}{1-0.64}\right) = 10.56.
    \end{align*}
    
    \item 
    Since $\operatorname{Var}[e_t] = 3$, we have the ACGF $G: \mathbb{C} \rightarrow \mathbb{C}$ defined as
    \begin{align*}
      G(z) &= 3(1+\sum_{j=1}^{\infty}1.2 \times 0.8^{j-1} z^j)(1+\sum_{j=1}^{\infty}1.2 \times 0.8^{j-1} z^{-j})\\
      &= 3\times\Bigg[\left(1+1.44\times \sum_{j=1}^{\infty}0.8^{2(j-1)}\right)
      + \left(1.2 + 1.44 \times \sum_{j=1}^{\infty}0.8^{2j-1}\right)z\\
      & \qquad \quad +\left(0.96 + 1.44 \times \sum_{j=1}^{\infty}0.8^{2j}\right)z^2
      +\cdots \Bigg].
    \end{align*}
    By the ACGF, 
    \begin{align*}
      \gamma(0) &= 15\\
      \gamma(1) &= 13.2\\
      \gamma(2) &= 10.56.
    \end{align*}
    This verifies the results in part (b).
   \end{enumerate}

\section*{Question 4}
\begin{enumerate}[label=\alph*.]
   \item The expectation is
   \begin{align*}
      \mathbb{E}[x_t] &= \mathbb{E}[\alpha] \cos t + \mathbb{E}[\beta] \sin t\\
      & = 0.
   \end{align*}
   The variance is
   \begin{align*}
      \operatorname{Var}[x_t] &= \operatorname{Var}[\alpha] \cos^2 t + \operatorname{Var}[\beta] \sin^2 t + 2 \operatorname{Cov}(\alpha, \beta) \cos t \sin t\\
      & = \cos^2 t + \sin^2 t\\
      & = 1.
   \end{align*}
   The second line is by the independence of $\alpha$ and $\beta$.

   \item No. Note that
   \begin{align*}
      \mathbb{E}[\alpha^2] &= \operatorname{Var}[\alpha] + \left(\mathbb{E}[\alpha]\right)^2\\
      &= 1\\
      \mathbb{E}[\beta^2] &= \operatorname{Var}[\beta] + \left(\mathbb{E}[\beta]\right)^2\\
      &= 1\\
      \mathbb{E}[\alpha \beta] &= \operatorname{Cov}(\alpha, \beta) + \mathbb{E}[\alpha] \mathbb{E}[\beta]\\
      &= 0.
   \end{align*}
   As a result, we have
   \begin{align*}
      \mathbb{E}[x_t x_{t-k}] &= \mathbb{E}[\alpha^2] \cos t \cos (t-k) + \mathbb{E}[\beta^2] \sin t \sin (t-k)\\
      &\qquad \quad + \mathbb{E}[\alpha \beta] [\cos t \sin (t-k) + \sin t \cos (t-k)]\\
      &= \cos t \cos (t-k) + \sin t \sin (t-k)\\
      &= \cos [t-(t-k)]\\
      &= \cos k.
   \end{align*}
\end{enumerate}
